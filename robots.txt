# 구포국수체험관 robots.txt
#
# 주의. robots 규약상 봇은 "자기 이름이 있는 그룹 하나"만 읽는다.
# 예전에는 Disallow가 `User-agent: *` 그룹에만 있고 Yeti·AI 크롤러는
# 각자 `Allow: /` 한 줄짜리 그룹이라, 그 봇들에게는 차단이 존재하지 않았다.
# 그래서 User-agent를 한 그룹에 이어 적어 같은 규칙이 모두에게 걸리게 한다.
# (연속된 User-agent 줄은 하나의 그룹으로 취급된다)

Disallow: /admin
Disallow: /admin.html
